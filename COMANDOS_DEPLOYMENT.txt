# ───────────────────────────────────────────────────────────────
# OPCIÓN B: DEPLOYMENT MANUAL (PASO A PASO)
# ───────────────────────────────────────────────────────────────

# 1. Modo mantenimiento
php artisan down --message="Actualizando sistema" --retry=60

# 2. Pull de cambios
git pull origin main

# 3. Instalar dependencias
composer install --no-dev --optimize-autoloader

# 4. Crear directorios necesarios
mkdir -p public/images/landing
chmod -R 775 public/images/landing

# 5. Limpiar TODA la caché
php artisan cache:clear
php artisan view:clear
php artisan config:clear
php artisan route:clear

# 6. Cachear para producción
php artisan config:cache
php artisan route:cache
php artisan view:cache

# 7. Optimizar autoload
composer dump-autoload -o

# 8. Configurar permisos
chmod -R 775 storage bootstrap/cache

# Si usas www-data (descomentar si aplica):
# chown -R www-data:www-data storage bootstrap/cache

# 9. Activar el sitio
php artisan up
